          content:
            application/json:
              schema:
                $ref: '#/components/schemas/Error400'
        '401':
          description: Não autorizado.
          
    get:
      tags:
        - Caronas (Rides)
      summary: Lista e busca caronas disponíveis com filtros
      description: Retorna uma lista paginada de caronas baseada em filtros de busca.
      security:
        - BearerAuth: []
      parameters:
        - $ref: '#/components/parameters/page'
        - $ref: '#/components/parameters/limit'
        - name: destination
          in: query
          required: false
          schema:
            type: string
          description: Filtra por destino (busca parcial).
        - name: origin
          in: query
          required: false
          schema:
            type: string
          description: Filtra por origem (busca parcial).
        - name: date
          in: query
          required: false
          schema:
            type: string
            format: date
          description: Filtra por data de partida (formato YYYY-MM-DD).
        - name: maxCost
          in: query
          required: false
          schema:
            type: number
            format: float
          description: Custo máximo da carona por passageiro.
        - name: orderBy
          in: query
          required: false
          schema:
            type: string
            enum: [price, departureTime]
          description: Campo para ordenação dos resultados.
        - name: orderDirection
          in: query
          required: false
          schema:
            type: string
            enum: [asc, desc]
          description: Direção da ordenação (ascendente ou descendente).
        - name: status
          in: query
          required: false
          schema:
            type: string
            enum: [PENDING, ACTIVE, IN_PROGRESS, COMPLETED, CANCELLED]
          description: Filtra por status da carona.
      responses:
        '200':
          description: Lista de caronas retornada com sucesso.
          content:
            application/json:
              schema:
                type: array
                items:
                  $ref: '#/components/schemas/Ride'
        '404':
          description: Nenhuma carona encontrada com os filtros fornecidos.
        '500':
          description: Erro interno do servidor.
