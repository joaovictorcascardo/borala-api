info:
  title: Borala API - Sistema de Caronas
  description: |
    API completa para gestão de caronas, reservas, veículos, usuários, avaliações e eventos.

    Possui autenticação via JWT e segue rigorosamente o padrão **snake_case** em todos os payloads,
    alinhado aos validadores Zod usados na API.

    Esta documentação foi aprimorada para um nível profissional, incluindo:
    - Schemas reutilizáveis
    - Exemplos detalhados
    - Respostas documentadas
    - Códigos de erro padronizados

  /rides:
    get:
      tags: [Caronas]

        - $ref: '#/components/parameters/Limit'
        - name: origin
          in: query
          schema: { type: string }
          description: Filtro de origem (parcial)
        - name: destination
          in: query
          schema: { type: string }
          description: Filtro de destino (parcial)
        - name: date
          in: query
          schema: { type: string, format: date }
          description: Filtro de data (YYYY-MM-DD)
        - name: maxCost
          in: query

          description: Custo máximo total estimado
        - name: status
          in: query

            type: string

            enum: [asc, desc]
            default: asc
          description: Direção (ascendente/descendente)

    ############################################################
    # USUÁRIOS
    ############################################################

    User:
      type: object
      properties:
        id: { type: integer }
        name: { type: string }
        email: { type: string }
        birth_date: { type: string, format: date }
        phone: { type: number }
        bio: { type: string, nullable: true }
        avatar_url: { type: string, nullable: true }

    UserPublic:
      type: object
      properties:
        id: { type: integer }
        name: { type: string }
        bio: { type: string, nullable: true }
        avatar_url: { type: string, nullable: true }

    UserCreateRequest:
      type: object
      required:
        - name
        - email
        - password
        - birth_date
        - phone
      properties:
        name: { type: string }
        email: { type: string }
        password: { type: string }
        birth_date: { type: string, format: date }
        phone: { type: number }
      example:
        name: "João Victor"
        email: "joao@email.com"
        password: "123456"
        birth_date: "2003-05-18"
        phone: 31999998888

    UserUpdateRequest:
      type: object
      properties:
        name: { type: string }
        phone: { type: number }
        bio: { type: string, nullable: true }
      example:
        name: "João Atualizado"
        phone: 31999995555
        bio: "Motorista há 3 anos"

    ############################################################
    # LOGIN
    ############################################################

    LoginRequest:
      type: object
      required: [email, password]
      properties:
        email: { type: string }
        password: { type: string }

    LoginResponse:
      type: object
      properties:
        token: { type: string }
        user:
          $ref: '#/components/schemas/User'
      example:
        token: "jwt_token_aqui"
        user:
          id: 1
          name: "João"
          email: "joao@email.com"

    ForgotPasswordRequest:
      type: object
      required: [email]
      properties:
        email: { type: string }

      type: object
      required: [token, password, password_confirmation]
      properties:
        token: { type: string }
        password: { type: string }
        password_confirmation: { type: string }

        destination_latitude: { type: number }
        destination_longitude: { type: number }
        departure_time: { type: string, format: date-time }
        available_seats: { type: integer }
